<objective>
Создать backend API для Quran Reader: эндпоинты для чтения аятов с пословной разбивкой,
добавления слов в очередь изучения, удаления слов из очереди.

Purpose: Frontend ридер получает данные из API — суры, аяты с позициями слов, флаг is_in_study.
Output: 4 новых эндпоинта под /api/v1/reader/*, все тесты зелёные.
</objective>

<context>
Существующие модели:
- Surah: id, number, name_arabic, name_english, name_transliteration, total_ayahs
- Ayah: id, surah_number, ayah_number, arabic_text, russian_translation
- Word: id, arabic, arabic_clean, frequency
- WordOccurrence: word_id, ayah_id, position (0-based позиция слова в аяте)
- UserCardProgress: user_id, word_id, next_review_date (наличие = слово в изучении)

extract_words() из app.utils.arabic разбивает arabic_text по пробелам (та же логика что в seeder).
Позиция в WordOccurrence соответствует индексу в extract_words(arabic_text).

Для каждого аята нужно:
1. Разбить arabic_text на слова через extract_words()
2. Для каждой позиции найти WordOccurrence → Word
3. Проверить есть ли UserCardProgress для текущего пользователя

Важно: не все позиции могут иметь WordOccurrence (частицы, союзы которые не вошли в словарь).
В этом случае word_id = null.

Enqueue: создать UserCardProgress(user_id, word_id, easiness_factor=2.5, interval=0,
repetitions=0, next_review_date=today) если не существует.

Dequeue: удалить UserCardProgress для данного пользователя и слова.
</context>

```python
from pydantic import BaseModel
from typing import Optional


class SurahInfo(BaseModel):
    number: int
    name_arabic: str
    name_english: str
    name_transliteration: Optional[str] = None
    total_ayahs: int

    model_config = {"from_attributes": True}


class WordInAyah(BaseModel):
    word_id: Optional[int]       # None если слово не в словаре
    position: int                 # позиция в аяте (0-based)
    arabic: str                   # слово с огласовками как в тексте
    is_in_study: bool = False     # True если UserCardProgress существует


class AyahWithWords(BaseModel):
    id: int
    surah_number: int
    ayah_number: int
    arabic_text: str
    russian_translation: Optional[str]
    words: list[WordInAyah]

    model_config = {"from_attributes": True}


class EnqueueRequest(BaseModel):
    word_ids: list[int]


class EnqueueResponse(BaseModel):
    added: int
    already_studying: int
```

Создать backend/app/api/v1/reader.py:
```python
from datetime import date
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session

from app.api.deps import get_current_user
from app.core.database import get_db
from app.models.user import User
from app.models.surah import Surah
from app.models.ayah import Ayah
from app.models.word import Word, WordOccurrence
from app.models.flashcard import UserCardProgress
from app.schemas.reader import (
    SurahInfo, AyahWithWords, WordInAyah,
    EnqueueRequest, EnqueueResponse,
)
from app.utils.arabic import extract_words

router = APIRouter(prefix="/reader", tags=["reader"])

DEFAULT_EF = 2.5


@router.get("/surahs", response_model=list[SurahInfo])
def get_surahs(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Список сур 30-го джуза (в порядке убывания номера — 114..78)."""
    surahs = (
        db.query(Surah)
        .order_by(Surah.number.asc())
        .all()
    )
    return surahs


@router.get("/ayahs/{surah_number}", response_model=list[AyahWithWords])
def get_ayahs(
    surah_number: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Аяты суры с пословной разбивкой и флагом is_in_study."""
    surah = db.query(Surah).filter(Surah.number == surah_number).first()
    if not surah:
        raise HTTPException(status_code=404, detail="Surah not found")

    ayahs = (
        db.query(Ayah)
        .filter(Ayah.surah_number == surah_number)
        .order_by(Ayah.ayah_number)
        .all()
    )

    # Загрузить все word_ids которые пользователь уже изучает
    studying_word_ids = set(
        row[0]
        for row in db.query(UserCardProgress.word_id)
        .filter(UserCardProgress.user_id == current_user.id)
        .all()
    )

    # Загрузить все WordOccurrence для аятов этой суры одним запросом
    ayah_ids = [a.id for a in ayahs]
    occurrences = (
        db.query(WordOccurrence)
        .filter(WordOccurrence.ayah_id.in_(ayah_ids))
        .all()
    )
    # Индекс: ayah_id → {position → (word_id, arabic)}
    occ_index: dict[int, dict[int, tuple[int, str]]] = {}
    for occ in occurrences:
        if occ.ayah_id not in occ_index:
            occ_index[occ.ayah_id] = {}
        word = db.query(Word).filter(Word.id == occ.word_id).first()
        if word:
            occ_index[occ.ayah_id][occ.position] = (word.id, word.arabic)

    result = []
    for ayah in ayahs:
        tokens = extract_words(ayah.arabic_text)
        words_in_ayah = []
        for pos, token in enumerate(tokens):
            if ayah.id in occ_index and pos in occ_index[ayah.id]:
                word_id, arabic = occ_index[ayah.id][pos]
                words_in_ayah.append(WordInAyah(
                    word_id=word_id,
                    position=pos,
                    arabic=arabic,
                    is_in_study=word_id in studying_word_ids,
                ))
            else:
                # Слово не в словаре (союз, частица и т.д.)
                words_in_ayah.append(WordInAyah(
                    word_id=None,
                    position=pos,
                    arabic=token,
                    is_in_study=False,
                ))
        result.append(AyahWithWords(
            id=ayah.id,
            surah_number=ayah.surah_number,
            ayah_number=ayah.ayah_number,
            arabic_text=ayah.arabic_text,
            russian_translation=ayah.russian_translation,
            words=words_in_ayah,
        ))

    return result


@router.post("/enqueue", response_model=EnqueueResponse)
def enqueue_words(
    req: EnqueueRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Добавить слова в очередь изучения (создать UserCardProgress)."""
    added = 0
    already_studying = 0
    today = date.today()

    for word_id in req.word_ids:
        # Проверить что слово существует
        word = db.query(Word).filter(Word.id == word_id).first()
        if not word:
            continue

        existing = (
            db.query(UserCardProgress)
            .filter(
                UserCardProgress.user_id == current_user.id,
                UserCardProgress.word_id == word_id,
            )
            .first()
        )
        if existing:
            already_studying += 1
            continue

        progress = UserCardProgress(
            user_id=current_user.id,
            word_id=word_id,
            easiness_factor=DEFAULT_EF,
            interval=0,
            repetitions=0,
            next_review_date=today,
        )
        db.add(progress)
        added += 1

    db.commit()
    return EnqueueResponse(added=added, already_studying=already_studying)


@router.delete("/words/{word_id}", status_code=status.HTTP_204_NO_CONTENT)
def dequeue_word(
    word_id: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Убрать слово из очереди изучения."""
    progress = (
        db.query(UserCardProgress)
        .filter(
            UserCardProgress.user_id == current_user.id,
            UserCardProgress.word_id == word_id,
        )
        .first()
    )
    if progress:
        db.delete(progress)
        db.commit()
```
